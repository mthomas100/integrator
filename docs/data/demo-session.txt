def split_line(line, delimiter=","):
    """Split one line, keeping delimiters that sit inside double quotes."""
    fields, field, quoted = [], "", False
    for ch in line:
        if ch == '"':
            quoted = not quoted
        elif ch == delimiter and not quoted:
            fields.append(field)
            field = ""
        else:
            field += ch
    fields.append(field)
    return fields
